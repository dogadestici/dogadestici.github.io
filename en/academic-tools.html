arXiv Subscriptions & Custom Python Mail Filter

To keep up with the latest literature in theoretical physics and general relativity, I subscribe to daily arXiv email updates. arXiv provides automated daily mailings that list all newly submitted preprints in specific subject categories.

How to Subscribe to Daily arXiv Updates

You can receive daily digests of new papers sent directly to your inbox by following these steps:

  • Create an Account / Log In: Visit the arXiv User Login page and sign in to your account.
  • Subscribe to Categories: Go to the arXiv Subscriptions Management page. Select your fields of interest—such as Mathematical Physics (math-ph) and General Relativity and Quantum Cosmology (gr-qc)—and enable email notifications.
  • Daily Digest: Every weekday, arXiv sends a structured email containing titles, abstracts, authors, and direct paper links submitted in your chosen categories.

Automated Python Email Filter

Since daily digests often contain dozens of preprints outside my immediate research scope, I use a custom Python script to filter the email content locally. By simply pasting the raw email text into VS Code, the script categorizes papers based on research interests and outputs titles, authors, and links. 

import re

INTERESTS = {
    "category_1": [
        "keyword_1", "keyword_2", "keyword_3"
    ],
    "category_2": [
        "keyword_1", "keyword_2", "keyword_3"
    ],
    "category_3": [
        "keyword_1", "keyword_2", "keyword_3"
    ],
    "category_4": [
        "keyword_1", "keyword_2", "keyword_3"
    ]
}

def analyze_expanded_arxiv_mail(raw_text):
    raw_papers = re.split(r'-{10,}', raw_text)
    
    categorized_papers = {cat: [] for cat in INTERESTS}
    categorized_papers["Other (Outside Interest)"] = []

    for block in raw_papers:
        if not block.strip() or "Title:" not in block:
            continue
            
        title_match = re.search(r'Title:\s*(.*?)(?=\n(?:Authors:|Categories:|Comments:|Report-no:)|$)', block, re.DOTALL)
        authors_match = re.search(r'Authors:\s*(.*?)(?=\n(?:Comments:|Categories:|MSC-class:|Report-no:|\\)|$)', block, re.DOTALL)
        
        title = title_match.group(1).replace('\n', ' ').strip() if title_match else "Unknown Title"
        authors_raw = authors_match.group(1).replace('\n', ' ').strip() if authors_match else "Unknown Authors"
        
        link_match = re.search(r'https://arxiv\.org/abs/\d{4}\.\d{4,5}', block)
        
        if link_match:
            paper_link = link_match.group(0)
        else:
            id_match = re.search(r'arXiv:\d{4}\.\d{4,5}', block)
            paper_link = f"https://arxiv.org/abs/{id_match.group(0).split(':')[1]}" if id_match else "Link Not Found"
        
        authors_cleaned = re.sub(r'\(.*?\)', '', authors_raw)
        clean_authors = [a.strip() for a in re.split(r',| and ', authors_cleaned) if a.strip()]
        
        block_lower = block.lower()
        matched_any_category = False
        
        for category, keywords in INTERESTS.items():
            if any(kw in block_lower for kw in keywords):
                categorized_papers[category].append({
                    "title": title,
                    "authors": clean_authors,
                    "link": paper_link
                })
                matched_any_category = True
                break
                
        if not matched_any_category:
            categorized_papers["Other (Outside Interest)"].append({
                "title": title,
                "authors": clean_authors,
                "link": paper_link
            })
            
    return categorized_papers

def print_results(categorized_data):
    for category, papers in categorized_data.items():
        if not papers:
            continue
        print(f"\n=== {category} ({len(papers)} Papers) ===")
        for idx, paper in enumerate(papers, 1):
            authors_str = ", ".join(paper['authors'])
            print(f"{idx}. Title: {paper['title']}")
            print(f"   Authors: {authors_str}")
            print(f"   Link: {paper['link']}")
            print("-" * 40)

email_content = r''' 
\\

'''

results = analyze_expanded_arxiv_mail(email_content)
print_results(results)
